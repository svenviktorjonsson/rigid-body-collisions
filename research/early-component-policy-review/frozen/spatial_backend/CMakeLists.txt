  set(LAPACK_LIBRARIES ${SPATIAL_LAPACK_RUNTIME} ${SPATIAL_BLAS_RUNTIME})
 endif()
 message(STATUS "Difficult friction recovery LAPACK: ${LAPACK_LIBRARIES}")
