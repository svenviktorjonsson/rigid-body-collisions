cmake_minimum_required(VERSION 3.22)
project(spatial_backend LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(USE_DOUBLE_PRECISION ON CACHE BOOL "" FORCE)
set(BUILD_BULLET2_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_CPU_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_OPENGL3_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET3 OFF CACHE BOOL "" FORCE)
set(BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_PYBULLET OFF CACHE BOOL "" FORCE)
set(INSTALL_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(bullet URL https://codeload.github.com/bulletphysics/bullet3/tar.gz/2c204c49e56ed15ec5fcfa71d199ab6d6570b3f5 URL_HASH SHA256=f7c486f10b9f645d0577249d02855e076e8368babaa3d31ced6095e7162baee9 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3 URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(bullet json)
add_executable(spatial_runner runner.cpp)
target_compile_features(spatial_runner PRIVATE cxx_std_17)
target_compile_definitions(spatial_runner PRIVATE BT_USE_DOUBLE_PRECISION)
target_include_directories(spatial_runner PRIVATE ${bullet_SOURCE_DIR}/src)
target_link_libraries(spatial_runner PRIVATE BulletDynamics BulletCollision LinearMath nlohmann_json::nlohmann_json)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(spatial_runner PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_executable(spatial_qp_checks qp_checks.cpp)
target_compile_features(spatial_qp_checks PRIVATE cxx_std_17)
target_compile_definitions(spatial_qp_checks PRIVATE BT_USE_DOUBLE_PRECISION)
target_include_directories(spatial_qp_checks PRIVATE ${bullet_SOURCE_DIR}/src)
target_link_libraries(spatial_qp_checks PRIVATE BulletDynamics BulletCollision LinearMath)
